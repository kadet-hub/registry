{{- /* CAT-37c: the prerequisites of the recipe named ., then those of the recipes it includes and of each in turn, from the READMEs and init.yml files catalog/build copied; a recipe without a page ends its branch. */ -}} {{- $name := . -}} {{- $all := slice -}} {{- with resources.Get (printf "classes/recipes/%s/README.md" $name) -}} {{- with findRESubmatch "(?m)^Requires: (.+)$" .Content 1 -}} {{- range findRESubmatch "`([^`]+)`" (index (index . 0) 1) -1 }}{{ $all = $all | append (index . 1) }}{{ end -}} {{- end -}} {{- end -}} {{- with resources.Get (printf "classes/recipes/%s/init.yml" $name) -}} {{- range (transform.Unmarshal .Content).classes -}} {{- with findRESubmatch `^recipes\.([a-z0-9-]+)$` . 1 -}} {{- range partial "recipe-requires.html" (index (index . 0) 1) }}{{ if not (in $all .) }}{{ $all = $all | append . }}{{ end }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- range $all -}} {{- range partial "recipe-requires.html" . }}{{ if not (in $all .) }}{{ $all = $all | append . }}{{ end }}{{ end -}} {{- end -}} {{- return $all -}}
