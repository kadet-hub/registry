{{- /* CAT-39: the image security report of the version a page shows, as {grade, class, text, full}. dict: report (generators or recipes), name, version. */ -}} {{- $s := site.Data.security -}} {{- $p := index (index $s .report) .name -}} {{- $r := dict "grade" "" "class" "" "full" "" -}} {{- if not .version -}} {{- $r = merge $r (dict "text" "Not scanned: no version is listed.") -}} {{- else if not $p -}} {{- $why := "no image scan was available when this page was built" -}} {{- with $s.scanned }}{{ $why = printf "the image scan of %s did not cover version %s" (substr . 0 10) $.version }}{{ end -}} {{- $r = merge $r (dict "text" (printf "Not scanned: %s." $why)) -}} {{- else if not $p.images -}} {{- $r = merge $r (dict "text" (printf "Version %s pins no image by digest." .version)) -}} {{- else -}} {{- $at := printf "%s %s UTC" (substr $s.scanned 0 10) (substr $s.scanned 11 5) -}} {{- $n := len $p.images -}} {{- $images := printf "%d image%s of version %s" $n (cond (eq $n 1) "" "s") .version -}} {{- $r = merge $r (dict "full" (printf "/security/%s/%s/" .report .name)) -}} {{- if $p.failed -}} {{- $r = merge $r (dict "text" (printf "Not graded: %d of the %s could not be scanned on %s." (int $p.failed) $images $at)) -}} {{- else -}} {{- $c := $p.counts -}} {{- $r = merge $r (dict "grade" $p.grade "class" (cond (eq $p.grade "-") "u" (lower $p.grade)) "text" (printf "%d finding%s, one per advisory and package, in the %s: CRITICAL %d, HIGH %d, MEDIUM %d, LOW %d, UNKNOWN %d. Scanned %s." (int $p.total) (cond (eq (int $p.total) 1) "" "s") $images (int (index $c "CRITICAL")) (int (index $c "HIGH")) (int (index $c "MEDIUM")) (int (index $c "LOW")) (int (index $c "UNKNOWN")) $at)) -}} {{- end -}} {{- end -}} {{- return $r -}}
