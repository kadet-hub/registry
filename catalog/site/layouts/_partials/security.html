(strings.FirstUpper $of))) -}} {{- else -}} {{- $at := printf "%s %s UTC" (substr $s.scanned 0 10) (substr $s.scanned 11 5) -}} {{- $n := int $p.count -}} {{- $images := printf "%d image%s of %s" $n (cond (eq $n 1) "" "s") $of -}} {{- with int $p.tagged }}{{ $images = printf "%s, %d by tag, digest at scan time" $images . }}{{ end -}} {{- if $p.failed -}} {{- $r = merge $r (dict "text" (printf "Not graded: %d of the %s could not be scanned on %s." (int $p.failed) $images $at)) -}} {{- else -}} {{- $c := $p.counts -}} {{- $r = merge $r (dict "grade" $p.grade "class" (cond (eq $p.grade "-") "u" (lower $p.grade)) "text" (printf "%d finding%s, one per advisory and package, in the %s: CRITICAL %d, HIGH %d, MEDIUM %d, LOW %d, UNKNOWN %d. Scanned %s." (int $p.total) (cond (eq (int $p.total) 1) "" "s") $images (int (index $c "CRITICAL")) (int (index $c "HIGH")) (int (index $c "MEDIUM")) (int (index $c "LOW")) (int (index $c "UNKNOWN")) $at)) -}} {{- end -}} {{- end -}} {{- return $r -}}
